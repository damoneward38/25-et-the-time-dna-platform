/* ── Time vortex scene ───────────────────────────────────── */
function TimeVortexScene() {
  const [currentLayer, setCurrentLayer] = useState(0);
  const [spiralAngle,  setSpiralAngle]  = useState(0);

  useEffect(() => {
    const iv = setInterval(() => {
      setSpiralAngle(p => (p + 0.5) % 360);
      setCurrentLayer(p => (p + 1) % TIME_LAYERS.length);
    }, 60);
    return () => clearInterval(iv);
  }, []);

  return (
    <div className="absolute inset-0 flex items-center justify-center
                    pointer-events-none overflow-hidden">

      {/* Deep space nebula */}
      {[
        { x:"50%", y:"50%", c:"rgba(99,102,241,0.08)",  w:800 },
        { x:"30%", y:"40%", c:"rgba(168,85,247,0.06)",  w:600 },
        { x:"70%", y:"60%", c:"rgba(6,182,212,0.05)",   w:500 },
        { x:"20%", y:"70%", c:"rgba(251,191,36,0.04)",  w:400 },
      ].map((b,i) => (
        <div key={i} className="absolute rounded-full" style={{
          width:b.w, height:b.w*0.65,
          left:b.x, top:b.y,
          background:b.c, filter:"blur(100px)",
          transform:"translate(-50%,-50%)",
        }} />
      ))}

      {/* Stars */}
      {Array.from({length:180},(_,i) => (
        <motion.div key={i} className="absolute rounded-full"
          style={{
            left:`${Math.random()*100}%`,
            top:`${Math.random()*100}%`,
            width:Math.random()*2.5+0.5,
            height:Math.random()*2.5+0.5,
            background:["#6366f1","#a855f7","#06b6d4","#fbbf24","#f0abfc","#fff"][i%6],
            opacity:Math.random()*0.5+0.1,
          }}
          animate={{opacity:[0.1,0.7,0.1],scale:[1,1.5,1]}}
          transition={{duration:2+Math.random()*3,delay:Math.random()*5,repeat:Infinity}}
        />
      ))}

      {/* DNA double helix time spiral */}
      <div className="absolute inset-0 flex items-center justify-center">

        {/* Outer time rings — each represents an era */}
        {TIME_LAYERS.map((layer,i) => {
          const size = 80 + i * 65;
          const isActive = i === currentLayer % TIME_LAYERS.length;
          return (
            <motion.div
              key={i}
              className="absolute rounded-full"
              style={{
                width:     size,
                height:    size * 0.35,
                border:    `1px ${isActive ? "solid" : "dashed"} ${layer.color}${isActive ? "60" : "20"}`,
                boxShadow: isActive ? `0 0 20px ${layer.color}30` : "none",
              }}
              animate={{
                rotate:       i % 2 === 0 ? 360 : -360,
                borderColor:  isActive ? `${layer.color}` : `${layer.color}20`,
              }}
              transition={{
                rotate:      { duration: 20 + i * 8, repeat: Infinity, ease: "linear" },
                borderColor: { duration: 0.5 },
              }}
            >
              {/* Era label on ring */}
              {isActive && (
                <div
                  className="absolute text-[8px] font-mono font-bold"
                  style={{
                    top:       -14,
                    left:      "50%",
                    transform: "translateX(-50%)",
                    color:     layer.color,
                    whiteSpace:"nowrap",
                  }}
                >
                  {layer.era}
                </div>
              )}
            </motion.div>
          );
        })}

        {/* DNA helix strands */}
        {Array.from({length:20},(_,i) => {
          const angle  = (i/20)*Math.PI*2 + (spiralAngle * Math.PI/180);
          const radius = 30;
          const yPos   = (i/20 - 0.5) * 400;
          const x1     = Math.cos(angle) * radius;
          const x2     = Math.cos(angle + Math.PI) * radius;
          const col1   = i%2===0 ? "#a855f7" : "#3b82f6";
          const col2   = i%2===0 ? "#3b82f6" : "#a855f7";
          return (
            <div key={i} className="absolute" style={{ top:`calc(50% + ${yPos}px)`, left:"50%" }}>
              {/* Left strand dot */}
              <motion.div
                className="absolute rounded-full"
                style={{
                  width:     6, height:6,
                  left:      x1-3, top:-3,
                  background: col1,
                  boxShadow:  `0 0 8px ${col1}`,
                }}
                animate={{ scale:[1,1.3,1] }}
                transition={{ duration:1, delay:i*0.05, repeat:Infinity }}
              />
              {/* Right strand dot */}
              <motion.div
                className="absolute rounded-full"
                style={{
                  width:     6, height:6,
                  left:      x2-3, top:-3,
                  background: col2,
                  boxShadow:  `0 0 8px ${col2}`,
                }}
                animate={{ scale:[1,1.3,1] }}
                transition={{ duration:1, delay:i*0.05+0.5, repeat:Infinity }}
              />
              {/* Connection */}
              <div
                className="absolute"
                style={{
                  width:     Math.abs(x1-x2),
                  height:    1,
                  left:      Math.min(x1,x2),
                  top:       -0.5,
                  background:`linear-gradient(90deg,${col2},${col1})`,
                  opacity:   0.4,
                }}
              />
            </div>
          );
        })}

        {/* Central time portal */}
        <div className="relative z-10 flex items-center justify-center">
          {/* Core pulse rings */}
          {[100,75,52,32,16].map((size,i) => (
            <motion.div
              key={i}
              className="absolute rounded-full"
              style={{
                width:     size, height:size,
                border:    `${i<2?2:1}px solid rgba(99,102,241,${0.5-i*0.08})`,
                boxShadow: `0 0 ${20-i*3}px rgba(99,102,241,${0.15-i*0.02})`,
              }}
              animate={{scale:[1,1.15,1],opacity:[0.8,0.3,0.8]}}
              transition={{duration:2.5+i*0.4,repeat:Infinity,delay:i*0.25}}
            />
          ))}

          {/* Core */}
          <motion.div
            className="relative z-10 rounded-2xl flex flex-col items-center justify-center"
            style={{
              width:      65, height:65,
              background: "radial-gradient(circle at 35% 35%, rgba(99,102,241,0.35), rgba(0,0,8,0.95))",
              border:     "1px solid rgba(99,102,241,0.5)",
              boxShadow:  "0 0 40px rgba(99,102,241,0.4), 0 0 80px rgba(168,85,247,0.2)",
            }}
            animate={{
              boxShadow:[
                "0 0 40px rgba(99,102,241,0.4), 0 0 80px rgba(168,85,247,0.2)",
                "0 0 80px rgba(99,102,241,0.7), 0 0 160px rgba(168,85,247,0.4)",
                "0 0 40px rgba(99,102,241,0.4), 0 0 80px rgba(168,85,247,0.2)",
              ],
            }}
            transition={{duration:3,repeat:Infinity}}
          >
            <span className="font-hebrew text-2xl font-black"
                  style={{color:"#6366f1",textShadow:"0 0 15px rgba(99,102,241,0.9)"}}>
              עֵת
            </span>
            <span className="text-[8px] font-mono text-white/30">TIME</span>
          </motion.div>
        </div>

        {/* Active era label */}
        <AnimatePresence mode="wait">
          <motion.div
            key={currentLayer}
            initial={{opacity:0,y:10}}
            animate={{opacity:1,y:0}}
            exit={{opacity:0,y:-10}}
            className="absolute glass rounded-2xl px-4 py-2 text-center"
            style={{
              bottom:      "15%",
              borderColor: `${TIME_LAYERS[currentLayer%TIME_LAYERS.length].color}30`,
            }}
          >
            <div className="text-xl">{TIME_LAYERS[currentLayer%TIME_LAYERS.length].emoji}</div>
            <div className="text-[10px] font-mono font-bold"
                 style={{color:TIME_LAYERS[currentLayer%TIME_LAYERS.length].color}}>
              {TIME_LAYERS[currentLayer%TIME_LAYERS.length].era}
            </div>
            <div className="text-[8px] text-white/30">
              {TIME_LAYERS[currentLayer%TIME_LAYERS.length].years}
            </div>
          </motion.div>
        </AnimatePresence>
      </div>

      {/* Data rain */}
      {Array.from({length:25},(_,i) => (
        <motion.div key={i}
          className="absolute font-mono text-[9px] pointer-events-none"
          style={{
            left:`${Math.random()*100}%`,
            color:["#6366f1","#a855f7","#fbbf24","#06b6d4","#f0abfc"][i%5],
            opacity:0.2,
          }}
          animate={{y:["-5vh","105vh"],opacity:[0,0.35,0]}}
          transition={{duration:4+Math.random()*5,delay:Math.random()*8,repeat:Infinity,ease:"linear"}}
        >
          {["DNA","528Hz","עֵת","TIME","∞","ψ","PAST","FUTURE","NOW","QUANTUM"][i%10]}
        </motion.div>
      ))}
    </div>
  );
}
